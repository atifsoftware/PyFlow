from core.controller import Controller

class ProductController(Controller):
    def index(self):
        # input রিডিং
        name = self.request.input("name")
        # ভিউ রেন্ডারিং
        return self.view("products.index", {"name": name})
        
    def api_data(self):
        # JSON রেসপন্স পাঠানো
        return self.json({"status": "success"})
